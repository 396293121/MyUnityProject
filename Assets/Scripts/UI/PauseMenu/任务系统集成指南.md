## 概述

本指南详细说明如何将新开发的任务系统与现有游戏项目集成，实现低耦合、高可维护性的任务管理功能。

#### 1.2 创建敌人击杀统计管理器

```csharp
/// <summary>
/// 敌人击杀统计管理器
/// 负责统计各种敌人的击杀数量，为任务系统提供数据支持
/// </summary>
public class EnemyKillTracker : MonoBehaviour
{
    public static EnemyKillTracker Instance { get; private set; }
    
    // 击杀统计字典
    private Dictionary<string, int> killCounts = new Dictionary<string, int>();
    
    // 击杀事件
    public event Action<string, int> OnEnemyKilled;
    
    void Awake()
    {
        if (Instance == null)
        {
            Instance = this;
            DontDestroyOnLoad(gameObject);
        }
        else
        {
            Destroy(gameObject);
        }
    }
    
    void Start()
    {
        // 注册所有敌人的死亡事件
        RegisterEnemyDeathEvents();
    }
    
    private void RegisterEnemyDeathEvents()
    {
        // 查找场景中所有敌人并注册死亡事件
        Enemy[] enemies = FindObjectsOfType<Enemy>();
        foreach (Enemy enemy in enemies)
        {
            enemy.OnDeath += OnEnemyDeath;
        }
    }
    
    private void OnEnemyDeath(Enemy enemy)
    {
        string enemyType = enemy.GetType().Name;
        
        if (!killCounts.ContainsKey(enemyType))
        {
            killCounts[enemyType] = 0;
        }
        
        killCounts[enemyType]++;
        
        // 触发击杀事件
        OnEnemyKilled?.Invoke(enemyType, killCounts[enemyType]);
        
        Debug.Log($"[EnemyKillTracker] 击杀 {enemyType}，总计: {killCounts[enemyType]}");
    }
    
    public int GetKillCount(string enemyType)
    {
        return killCounts.ContainsKey(enemyType) ? killCounts[enemyType] : 0;
    }
}
```

### 第三步：任务管理器扩展

#### 3.1 扩展QuestManager以支持击杀任务

```csharp
public partial class QuestManager : MonoBehaviour
{
    void Start()
    {
        // 原有初始化代码...
        
        // 注册击杀事件监听
        RegisterKillEventListeners();
    }
    
    private void RegisterKillEventListeners()
    {
        if (EnemyKillTracker.Instance != null)
        {
            EnemyKillTracker.Instance.OnEnemyKilled += OnEnemyKilled;
        }
    }
    
    private void OnEnemyKilled(string enemyType, int totalKills)
    {
        // 更新所有相关的击杀任务
        foreach (var quest in activeQuests)
        {
            UpdateKillQuest(quest, enemyType, totalKills);
        }
    }
    
    private void UpdateKillQuest(QuestData quest, string enemyType, int totalKills)
    {
        foreach (var objective in quest.objectives)
        {
            if (objective.type == QuestObjectiveType.Kill && 
                objective.targetId == enemyType && 
                !objective.isCompleted)
            {
                objective.currentCount = totalKills;
                objective.description = $"击杀{GetEnemyDisplayName(enemyType)} ({objective.currentCount}/{objective.targetCount})";
                
                if (objective.currentCount >= objective.targetCount)
                {
                    objective.isCompleted = true;
                    objective.description = $"击杀{GetEnemyDisplayName(enemyType)} (已完成)";
                }
                
                // 触发任务进度更新事件
                OnQuestProgressUpdated?.Invoke(quest);
                
                // 检查任务是否完成
                CheckQuestCompletion(quest);
                break;
            }
        }
    }
    
    private string GetEnemyDisplayName(string enemyType)
    {
        switch (enemyType)
        {
            case "WildBoar": return "野猪";
            default: return enemyType;
        }
    }
}
```

### 第四步：NPC任务分发集成

#### 4.1 扩展NPCController支持任务分发

```csharp
public partial class NPCController : MonoBehaviour
{
    [Header("任务系统")]
    [SerializeField] private List<QuestData> availableQuests = new List<QuestData>();
    [SerializeField] private bool isQuestGiver = false;
    
    private void Start()
    {
        // 原有初始化代码...
        
        // 初始化任务系统
        InitializeQuestSystem();
    }
    
    private void InitializeQuestSystem()
    {
        if (isQuestGiver && QuestManager.Instance != null)
        {
            // 注册任务完成事件
            QuestManager.Instance.OnQuestCompleted += OnQuestCompleted;
        }
    }
    
    private void OnQuestCompleted(QuestData quest)
    {
        // 当玩家完成任务时，可以在这里处理NPC的反应
        if (availableQuests.Contains(quest))
        {
            Debug.Log($"[NPCController] 玩家完成了任务: {quest.questName}");
        }
    }
    
    /// <summary>
    /// 给玩家分发任务
    /// </summary>
    public void GiveQuestToPlayer(string questId)
    {
        var quest = availableQuests.Find(q => q.questId == questId);
        if (quest != null && QuestManager.Instance != null)
        {
            QuestManager.Instance.StartQuest(quest);
            Debug.Log($"[NPCController] 分发任务: {quest.questName}");
        }
    }
    
    /// <summary>
    /// 检查是否有可用任务
    /// </summary>
    public bool HasAvailableQuests()
    {
        return isQuestGiver && availableQuests.Count > 0;
    }
}
```

### 第五步：对话系统集成

#### 5.1 创建任务对话命令

```csharp
/// <summary>
/// Fungus自定义命令：给予任务
/// </summary>
[CommandInfo("Quest", "Give Quest", "给玩家分发指定任务")]
public class GiveQuestCommand : Command
{
    [SerializeField] protected QuestData questToGive;
    
    public override void OnEnter()
    {
        if (questToGive != null && QuestManager.Instance != null)
        {
            QuestManager.Instance.StartQuest(questToGive);
            Debug.Log($"[GiveQuestCommand] 通过对话分发任务: {questToGive.questName}");
        }
        
        Continue();
    }
}

/// <summary>
/// Fungus自定义命令：检查任务状态
/// </summary>
[CommandInfo("Quest", "Check Quest", "检查指定任务的状态")]
public class CheckQuestCommand : Command
{
    [SerializeField] protected string questId;
    [SerializeField] protected BooleanVariable questCompleted;
    
    public override void OnEnter()
    {
        if (QuestManager.Instance != null)
        {
            bool isCompleted = QuestManager.Instance.IsQuestCompleted(questId);
            if (questCompleted != null)
            {
                questCompleted.Value = isCompleted;
            }
        }
        
        Continue();
    }
}
```

```csharp
/// <summary>
/// 任务通知管理器
/// </summary>
public class QuestNotificationManager : MonoBehaviour
{
    [Header("UI引用")]
    [SerializeField] private GameObject notificationPrefab;
    [SerializeField] private Transform notificationContainer;
    
    public static QuestNotificationManager Instance { get; private set; }
    
    void Awake()
    {
        if (Instance == null)
        {
            Instance = this;
            DontDestroyOnLoad(gameObject);
        }
        else
        {
            Destroy(gameObject);
        }
    }
    
    void Start()
    {
        // 注册任务事件
        if (QuestManager.Instance != null)
        {
            QuestManager.Instance.OnQuestStarted += ShowQuestStartNotification;
            QuestManager.Instance.OnQuestCompleted += ShowQuestCompleteNotification;
            QuestManager.Instance.OnQuestProgressUpdated += ShowQuestProgressNotification;
        }
    }
    
    private void ShowQuestStartNotification(QuestData quest)
    {
        ShowNotification($"新任务: {quest.questName}", Color.yellow);
    }
    
    private void ShowQuestCompleteNotification(QuestData quest)
    {
        ShowNotification($"任务完成: {quest.questName}", Color.green);
    }
    
    private void ShowQuestProgressNotification(QuestData quest)
    {
        var objective = quest.objectives.FirstOrDefault(o => !o.isCompleted);
        if (objective != null)
        {
            ShowNotification($"任务进度: {objective.description}", Color.cyan);
        }
    }
    
    private void ShowNotification(string message, Color color)
    {
        // 实现通知显示逻辑
        Debug.Log($"[QuestNotification] {message}");
    }
}
```

## 快速配置示例

### 创建"击杀10只野猪"任务

1. **创建任务数据**：
   - 在Project窗口右键 → Create → Quest → Kill Quest
   - 命名为"KillWildBoarQuest"
   - 配置目标敌人类型为"WildBoar"
   - 设置击杀数量为10

2. **配置任务给予NPC**：
   - 选择村长NPC
   - 在NPCController组件中勾选"Is Quest Giver"
   - 将创建的任务数据添加到"Available Quests"列表

3. **设置对话流程**：
   - 在Fungus Flowchart中创建对话块
   - 添加"Give Quest"命令
   - 配置任务数据引用

4. **测试任务系统**：
   - 与NPC对话接受任务
   - 击杀野猪查看进度更新
   - 完成任务获得奖励
